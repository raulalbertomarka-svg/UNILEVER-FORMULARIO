<script>
const PRODUCTOS_VALIDOS = Array.from(document.querySelectorAll("#lista-productos-datalist option")).map(opt => opt.value);

// ============================================================
// DATOS PDV: REEMPLAZAR con los relevadores y PDV de Unilever
// ============================================================
const datosPDV = {
  "RELEVADOR EJEMPLO 1": ["PDV EJEMPLO A", "PDV EJEMPLO B", "PDV EJEMPLO C"],
  "RELEVADOR EJEMPLO 2": ["PDV EJEMPLO D", "PDV EJEMPLO E"]
};

const todosLosRelevadores = Object.keys(datosPDV);

// ============================================================
// AUTOCOMPLETE PERSONALIZADO
// ============================================================
let highlightedIndex = -1;

function normalizarTexto(texto) {
  return texto.toLowerCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "");
}

function resaltarCoincidencia(nombre, query) {
  const nombreNorm = normalizarTexto(nombre);
  const queryNorm = normalizarTexto(query);
  const idx = nombreNorm.indexOf(queryNorm);
  if (idx === -1) return nombre;
  const antes = nombre.substring(0, idx);
  const coincide = nombre.substring(idx, idx + query.length);
  const despues = nombre.substring(idx + query.length);
  return `${antes}<span class="match-highlight">${coincide}</span>${despues}`;
}

function mostrarDropdown(resultados, query) {
  const dropdown = document.getElementById("vendedor-dropdown");
  highlightedIndex = -1;
  if (resultados.length === 0) {
    dropdown.innerHTML = `<div class="autocomplete-no-results">No se encontraron resultados para "${query}"</div>`;
  } else {
    dropdown.innerHTML = resultados.map((nombre, i) =>
      `<div class="autocomplete-item" data-valor="${nombre}" onmousedown="seleccionarRelevador('${nombre.replace(/'/g,"\\'")}')">
        ${resaltarCoincidencia(nombre, query)}
      </div>`
    ).join("");
  }
  dropdown.classList.add("visible");
}

function cerrarDropdown() {
  document.getElementById("vendedor-dropdown").classList.remove("visible");
  highlightedIndex = -1;
}

function onVendedorInput(input) {
  const query = input.value.trim();
  input.value = input.value.toUpperCase();
  if (query.length < 2) { cerrarDropdown(); actualizarPDV(); return; }
  const queryNorm = normalizarTexto(query);
  const resultados = todosLosRelevadores.filter(nombre => normalizarTexto(nombre).includes(queryNorm));
  mostrarDropdown(resultados, query);
  actualizarPDV();
}

function onVendedorFocus() {
  const query = document.getElementById("vendedor").value.trim();
  if (query.length >= 2) {
    const queryNorm = normalizarTexto(query);
    const resultados = todosLosRelevadores.filter(nombre => normalizarTexto(nombre).includes(queryNorm));
    mostrarDropdown(resultados, query);
  }
}

function onVendedorKeydown(event) {
  const dropdown = document.getElementById("vendedor-dropdown");
  const items = dropdown.querySelectorAll(".autocomplete-item");
  if (event.key === "ArrowDown") {
    event.preventDefault();
    highlightedIndex = Math.min(highlightedIndex + 1, items.length - 1);
    actualizarHighlight(items);
  } else if (event.key === "ArrowUp") {
    event.preventDefault();
    highlightedIndex = Math.max(highlightedIndex - 1, 0);
    actualizarHighlight(items);
  } else if (event.key === "Enter") {
    event.preventDefault();
    if (highlightedIndex >= 0 && items[highlightedIndex]) {
      seleccionarRelevador(items[highlightedIndex].dataset.valor);
    }
  } else if (event.key === "Escape") {
    cerrarDropdown();
  }
}

function actualizarHighlight(items) {
  items.forEach((item, i) => item.classList.toggle("highlighted", i === highlightedIndex));
  if (items[highlightedIndex]) items[highlightedIndex].scrollIntoView({ block: "nearest" });
}

function seleccionarRelevador(nombre) {
  document.getElementById("vendedor").value = nombre;
  cerrarDropdown();
  actualizarPDV();
}

document.addEventListener("click", function(e) {
  if (!e.target.closest(".autocomplete-wrapper")) {
    document.querySelectorAll(".autocomplete-dropdown").forEach(d => d.classList.remove("visible"));
    highlightedIndex = -1;
  }
});

function actualizarPDV() {
  const relevadorSeleccionado = document.getElementById("vendedor").value;
  const selectPDV = document.getElementById("pdv");
  if (datosPDV[relevadorSeleccionado]) {
    selectPDV.innerHTML = '<option value="">-- SELECCIONE PDV --</option>';
    datosPDV[relevadorSeleccionado].forEach(pdv => {
      let option = document.createElement("option");
      option.value = pdv;
      option.textContent = pdv;
      selectPDV.appendChild(option);
    });
  } else {
    selectPDV.innerHTML = '<option value="">-- RELEVADOR NO VÁLIDO --</option>';
  }
}

function cambiarTab(t) {
  document.getElementById('panel-vencidos').style.display = (t === 'venc') ? 'block' : 'none';
  document.getElementById('panel-proximos').style.display = (t === 'prox') ? 'block' : 'none';
  document.getElementById('panel-averiados').style.display = (t === 'aver') ? 'block' : 'none';
  document.getElementById('tab-venc').className = (t === 'venc') ? 'btn-tab active-venc' : 'btn-tab';
  document.getElementById('tab-prox').className = (t === 'prox') ? 'btn-tab active-prox' : 'btn-tab';
  document.getElementById('tab-aver').className = (t === 'aver') ? 'btn-tab active-aver' : 'btn-tab';
}

let respuestas = { venc: null, prox: null, aver: null };

function decidirCarga(tipo, tiene) {
  respuestas[tipo] = tiene;
  document.getElementById(`${tipo}-si`).className = tiene ? 'btn-sn active-si' : 'btn-sn';
  document.getElementById(`${tipo}-no`).className = !tiene ? 'btn-sn active-no' : 'btn-sn';
  const idSeccion = tipo === 'venc' ? 'vencidos' : (tipo === 'prox' ? 'proximos' : 'averiados');
  document.getElementById(`seccion-${idSeccion}`).style.display = tiene ? "block" : "none";
  const cont = document.getElementById(`cont-${idSeccion}`);
  if (tiene && cont.children.length === 0) agregarFila(tipo);
}

function validarProducto(input) {
  const errorMsg = input.parentElement.querySelector(".msg-error");
  if (input.value === "" || PRODUCTOS_VALIDOS.includes(input.value)) {
    input.classList.remove('input-error');
    if(errorMsg) errorMsg.style.display = "none";
  } else {
    input.classList.add('input-error');
    if(errorMsg) errorMsg.style.display = "block";
  }
}

// ============================================================
// AUTOCOMPLETE DE PRODUCTOS
// ============================================================
let prodHighlightIndex = {};

function onProductoInput(input, uid) {
  input.value = input.value.toUpperCase();
  const query = input.value.trim();
  if (query.length < 2) { cerrarDropdownProducto(uid); validarProducto(input); return; }
  const queryNorm = normalizarTexto(query);
  const resultados = PRODUCTOS_VALIDOS.filter(p => normalizarTexto(p).includes(queryNorm)).slice(0, 30);
  mostrarDropdownProducto(uid, resultados, query, input);
  validarProducto(input);
}

function onProductoFocus(input, uid) {
  const query = input.value.trim();
  if (query.length >= 2) {
    const queryNorm = normalizarTexto(query);
    const resultados = PRODUCTOS_VALIDOS.filter(p => normalizarTexto(p).includes(queryNorm)).slice(0, 30);
    mostrarDropdownProducto(uid, resultados, query, input);
  }
}

function mostrarDropdownProducto(uid, resultados, query, input) {
  const dropdown = document.getElementById('prod-drop-' + uid);
  prodHighlightIndex[uid] = -1;
  if (resultados.length === 0) {
    dropdown.innerHTML = '<div class="autocomplete-no-results">Sin resultados para "' + query + '"</div>';
  } else {
    dropdown.innerHTML = resultados.map(p =>
      '<div class="autocomplete-item" data-valor="' + p.replace(/"/g,'&quot;') + '" onmousedown="seleccionarProducto(this, \'' + uid + '\')">' + resaltarCoincidencia(p, query) + '</div>'
    ).join('');
  }
  dropdown.classList.add('visible');
}

function cerrarDropdownProducto(uid) {
  const dropdown = document.getElementById('prod-drop-' + uid);
  if (dropdown) dropdown.classList.remove('visible');
  prodHighlightIndex[uid] = -1;
}

function seleccionarProducto(item, uid) {
  const valor = item.dataset.valor;
  const dropdown = document.getElementById('prod-drop-' + uid);
  const wrapper = dropdown.closest('.autocomplete-wrapper');
  const input = wrapper.querySelector('.p-prod');
  input.value = valor;
  cerrarDropdownProducto(uid);
  validarProducto(input);
}

function onProductoKeydown(event, uid) {
  const dropdown = document.getElementById('prod-drop-' + uid);
  const items = dropdown.querySelectorAll('.autocomplete-item');
  if (prodHighlightIndex[uid] === undefined) prodHighlightIndex[uid] = -1;
  if (event.key === 'ArrowDown') {
    event.preventDefault();
    prodHighlightIndex[uid] = Math.min(prodHighlightIndex[uid] + 1, items.length - 1);
    items.forEach((it, i) => it.classList.toggle('highlighted', i === prodHighlightIndex[uid]));
    if (items[prodHighlightIndex[uid]]) items[prodHighlightIndex[uid]].scrollIntoView({ block: 'nearest' });
  } else if (event.key === 'ArrowUp') {
    event.preventDefault();
    prodHighlightIndex[uid] = Math.max(prodHighlightIndex[uid] - 1, 0);
    items.forEach((it, i) => it.classList.toggle('highlighted', i === prodHighlightIndex[uid]));
    if (items[prodHighlightIndex[uid]]) items[prodHighlightIndex[uid]].scrollIntoView({ block: 'nearest' });
  } else if (event.key === 'Enter') {
    event.preventDefault();
    if (prodHighlightIndex[uid] >= 0 && items[prodHighlightIndex[uid]]) {
      seleccionarProducto(items[prodHighlightIndex[uid]], uid);
    }
  } else if (event.key === 'Escape') {
    cerrarDropdownProducto(uid);
  }
}

function toggleOtros(checkbox) {
  const row = checkbox.closest(".producto-row");
  const inputOtros = row.querySelector(".input-otros-especificar");
  inputOtros.style.display = checkbox.checked ? "block" : "none";
  if(!checkbox.checked) inputOtros.value = "";
}

document.getElementById("formulario").onsubmit = function(e) {
  e.preventDefault();
  
  if (respuestas.venc === null || respuestas.prox === null || respuestas.aver === null) { 
    alert("DEBE COMPLETAR TODOS LOS APARTADOS (VENCIDOS, PRÓXIMOS Y AVERIADOS)."); 
    return; 
  }

  if (!datosPDV[document.getElementById("vendedor").value]) {
    alert("POR FAVOR SELECCIONE UN RELEVADOR VÁLIDO DE LA LISTA.");
    return;
  }

  let inputsProductos = document.querySelectorAll(".p-prod");
  let hayErrores = false;
  inputsProductos.forEach(input => {
    if (!PRODUCTOS_VALIDOS.includes(input.value)) {
      input.classList.add('input-error');
      const errorMsg = input.parentElement.querySelector(".msg-error");
      if(errorMsg) errorMsg.style.display = "block";
      hayErrores = true;
    }
  });

  if (hayErrores) {
    alert("❌ NO SE PUEDE ENVIAR: HAY PRODUCTOS NO VÁLIDOS EN LA LISTA.");
    return;
  }

  const btn = document.getElementById("btnEnvio");
  btn.disabled = true; btn.innerText = "ENVIANDO...";

  const payload = {
    relevador: document.getElementById("vendedor").value,
    pdv: document.getElementById("pdv").value,
    vencidos: [],
    proximos: [],
    averiados: []
  };

  document.querySelectorAll("#cont-vencidos .producto-row").forEach(r => {
    payload.vencidos.push({ 
      producto: r.querySelector(".p-prod").value, 
      extra: r.querySelector(".p-extra").value, 
      cant: r.querySelector(".p-cant").value 
    });
  });

  document.querySelectorAll("#cont-proximos .producto-row").forEach(r => {
    payload.proximos.push({ 
      producto: r.querySelector(".p-prod").value, 
      extra: r.querySelector(".p-extra").value, 
      cant: r.querySelector(".p-cant").value 
    });
  });

  document.querySelectorAll("#cont-averiados .producto-row").forEach(r => {
    let seleccionados = [];
    r.querySelectorAll(".checklist-container input[type='checkbox']:checked").forEach(check => {
      if(check.value === "OTROS") {
        let espec = r.querySelector(".input-otros-especificar").value;
        seleccionados.push("OTROS: " + (espec ? espec : "SIN ESPECIFICAR"));
      } else {
        seleccionados.push(check.value);
      }
    });

    if(seleccionados.length === 0) {
      alert("POR FAVOR SELECCIONE AL MENOS UN TIPO DE AVERÍA.");
      btn.disabled = false; btn.innerText = "ENVIAR REPORTE";
      throw "Validación Checklist fallida";
    }

    payload.averiados.push({ 
      producto: r.querySelector(".p-prod").value, 
      observacion: seleccionados.join(" | "), 
      cant: r.querySelector(".p-cant").value 
    });
  });

  // ⚠️ REEMPLAZAR con la URL del Google Apps Script de Unilever
  fetch("URL_DEL_GOOGLE_SCRIPT_DE_UNILEVER_AQUI", {
    method: "POST",
    mode: "no-cors",
    body: JSON.stringify(payload)
  }).then(() => { 
    alert("REPORTE ENVIADO CORRECTAMENTE A UNILEVER"); 
    location.reload(); 
  });
};
</script>
